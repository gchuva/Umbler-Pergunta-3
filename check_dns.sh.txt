#!/bin/bash

# ============================================================
# Script: check_dns_records.sh
# Verifica registros SPF, DKIM, DMARC e PTR/rDNS de um domínio
# ============================================================

# Verifica se o domínio foi passado como argumento
if [ -z "$1" ]; then
    echo "Uso: $0 <dominio>"
    echo "Exemplo: $0 exemplo.com"
    exit 1
fi

DOMINIO="$1"
DATA=$(date '+%Y-%m-%d %H:%M:%S')

# Cores para output
VERMELHO='\033[0;31m'
VERDE='\033[0;32m'
AMARELO='\033[1;33m'
AZUL='\033[0;34m'
NC='\033[0m' # No Color

# Função para imprimir cabeçalho
imprimir_cabecalho() {
    echo "============================================================"
    echo "  RELATÓRIO DE VERIFICAÇÃO DNS - $DOMINIO"
    echo "  Data: $DATA"
    echo "============================================================"
    echo ""
}

# Função para verificar se um comando existe
verificar_comando() {
    if ! command -v $1 &> /dev/null; then
        echo -e "${VERMELHO}Erro: $1 não está instalado.${NC}"
        exit 1
    fi
}

# Verificar se dig ou host estão disponíveis
if ! command -v dig &> /dev/null && ! command -v host &> /dev/null; then
    echo -e "${VERMELHO}Erro: Instale 'dig' (dnsutils) ou 'host' (bind-utils)${NC}"
    echo "Debian/Ubuntu: apt install dnsutils"
    echo "RHEL/CentOS: yum install bind-utils"
    exit 1
fi

# Usar dig se disponível, senão host
if command -v dig &> /dev/null; then
    DNS_TOOL="dig"
else
    DNS_TOOL="host"
fi

imprimir_cabecalho
verificar_comando $DNS_TOOL

echo -e "${AZUL}Ferramenta DNS utilizada: $DNS_TOOL${NC}"
echo ""

# ============================================================
# 1. VERIFICAÇÃO DE REGISTRO SPF
# ============================================================
echo "============================================================"
echo "1. VERIFICANDO REGISTRO SPF"
echo "============================================================"

if [ "$DNS_TOOL" = "dig" ]; then
    SPF_RECORD=$(dig +short TXT $DOMINIO | grep -i "spf" | head -1)
else
    SPF_RECORD=$(host -t TXT $DOMINIO | grep -i "spf" | head -1)
fi

if [ -n "$SPF_RECORD" ]; then
    echo -e "${VERDE}✓ SPF ENCONTRADO${NC}"
    echo "Registro: $SPF_RECORD"
    echo ""
    
    # Verificar se SPF tem sintaxe básica válida
    if echo "$SPF_RECORD" | grep -q "v=spf1"; then
        echo -e "${VERDE}  ✓ Versão SPF correta (v=spf1)${NC}"
    else
        echo -e "${VERMELHO}  ✗ Versão SPF incorreta (deve começar com v=spf1)${NC}"
    fi
    
    # Verificar se tem mecanismo 'all'
    if echo "$SPF_RECORD" | grep -qE "\ball\b"; then
        echo -e "${VERDE}  ✓ Mecanismo 'all' presente${NC}"
        
        # Verificar se é -all (fail) ou ~all (softfail)
        if echo "$SPF_RECORD" | grep -q "\-all"; then
            echo -e "${VERDE}  ✓ Política rigorosa (-all) - RECOMENDADO${NC}"
        elif echo "$SPF_RECORD" | grep -q "\~all"; then
            echo -e "${AMARELO}  ! Política branda (~all) - Considere usar -all${NC}"
        else
            echo -e "${AMARELO}  ! Política muito permissiva - Considere usar -all ou ~all${NC}"
        fi
    else
        echo -e "${VERMELHO}  ✗ Ausente mecanismo 'all' - SPF incompleto${NC}"
    fi
else
    echo -e "${VERMELHO}✗ SPF NÃO ENCONTRADO${NC}"
    echo "  Recomendação: Adicione um registro TXT com SPF para evitar spam"
    echo "  Exemplo: v=spf1 mx a ip4:SEU_IP ~all"
fi

echo ""

# ============================================================
# 2. VERIFICAÇÃO DE REGISTRO DMARC
# ============================================================
echo "============================================================"
echo "2. VERIFICANDO REGISTRO DMARC"
echo "============================================================"

DMARC_DOMINIO="_dmarc.$DOMINIO"

if [ "$DNS_TOOL" = "dig" ]; then
    DMARC_RECORD=$(dig +short TXT $DMARC_DOMINIO | grep -i "dmarc" | head -1)
else
    DMARC_RECORD=$(host -t TXT $DMARC_DOMINIO | grep -i "dmarc" | head -1)
fi

if [ -n "$DMARC_RECORD" ]; then
    echo -e "${VERDE}✓ DMARC ENCONTRADO${NC}"
    echo "Registro: $DMARC_RECORD"
    echo ""
    
    # Verificar versão DMARC
    if echo "$DMARC_RECORD" | grep -q "v=DMARC1"; then
        echo -e "${VERDE}  ✓ Versão DMARC correta (v=DMARC1)${NC}"
    else
        echo -e "${VERMELHO}  ✗ Versão DMARC incorreta${NC}"
    fi
    
    # Verificar política (p=)
    if echo "$DMARC_RECORD" | grep -qE "p=reject"; then
        echo -e "${VERDE}  ✓ Política rigorosa (p=reject) - RECOMENDADO${NC}"
    elif echo "$DMARC_RECORD" | grep -qE "p=quarantine"; then
        echo -e "${AMARELO}  ! Política de quarentena (p=quarantine)${NC}"
    elif echo "$DMARC_RECORD" | grep -qE "p=none"; then
        echo -e "${AMARELO}  ! Política de monitoramento (p=none) - Apenas coleta dados${NC}"
    else
        echo -e "${VERMELHO}  ✗ Política DMARC não definida${NC}"
    fi
    
    # Verificar se tem email de reporte (rua=)
    if echo "$DMARC_RECORD" | grep -qE "rua="; then
        echo -e "${VERDE}  ✓ Email de reporte agregado configurado (rua)${NC}"
    else
        echo -e "${AMARELO}  ! Sem email de reporte (rua) - Recomendado configurar${NC}"
    fi
    
else
    echo -e "${VERMELHO}✗ DMARC NÃO ENCONTRADO${NC}"
    echo "  Domínio verificado: $DMARC_DOMINIO"
    echo "  Recomendação: Adicione DMARC para proteger contra phishing"
    echo "  Exemplo: v=DMARC1; p=reject; rua=mailto:admin@$DOMINIO"
fi

echo ""

# ============================================================
# 3. VERIFICAÇÃO DE REGISTRO DKIM
# ============================================================
echo "============================================================"
echo "3. VERIFICANDO REGISTRO DKIM"
echo "============================================================"

# Seletores comuns de DKIM
SELETORES_DKIM=("default" "google" "selector1" "selector2" "mail" "dkim" "s1" "s2")
DKIM_ENCONTRADO=0

echo "Buscando seletores DKIM comuns..."
echo ""

for seletor in "${SELETORES_DKIM[@]}"; do
    DKIM_DOMINIO="${seletor}._domainkey.$DOMINIO"
    
    if [ "$DNS_TOOL" = "dig" ]; then
        DKIM_RECORD=$(dig +short TXT $DKIM_DOMINIO 2>/dev/null | head -1)
    else
        DKIM_RECORD=$(host -t TXT $DKIM_DOMINIO 2>/dev/null | grep -v "has no" | head -1)
    fi
    
    if [ -n "$DKIM_RECORD" ]; then
        echo -e "${VERDE}✓ DKIM ENCONTRADO (seletor: $seletor)${NC}"
        echo "  Domínio: $DKIM_DOMINIO"
        echo "  Registro: ${DKIM_RECORD:0:100}..."
        echo ""
        DKIM_ENCONTRADO=1
        
        # Verificar se tem chave pública (v=DKIM1)
        if echo "$DKIM_RECORD" | grep -q "v=DKIM1"; then
            echo -e "${VERDE}  ✓ Versão DKIM correta (v=DKIM1)${NC}"
        else
            echo -e "${AMARELO}  ! Versão DKIM não identificada${NC}"
        fi
        
        # Verificar se tem chave (k=)
        if echo "$DKIM_RECORD" | grep -qE "k=rsa"; then
            echo -e "${VERDE}  ✓ Algoritmo RSA configurado${NC}"
        fi
        
        break
    fi
done

if [ $DKIM_ENCONTRADO -eq 0 ]; then
    echo -e "${VERMELHO}✗ DKIM NÃO ENCONTRADO${NC}"
    echo "  Nenhum seletor DKIM comum encontrado"
    echo "  Seletores verificados: ${SELETORES_DKIM[*]}"
    echo ""
    echo "  Observação: O DKIM pode estar configurado com um seletor personalizado."
    echo "  Consulte seu provedor de email para obter o seletor correto."
    echo "  Recomendação: Configure DKIM para autenticar emails enviados"
fi

echo ""

# ============================================================
# 4. VERIFICAÇÃO DE PTR/rDNS (Reverse DNS)
# ============================================================
echo "============================================================"
echo "4. VERIFICANDO REGISTRO PTR/rDNS"
echo "============================================================"

# Obter IP do domínio
if [ "$DNS_TOOL" = "dig" ]; then
    IP_DOMINIO=$(dig +short $DOMINIO | head -1)
else
    IP_DOMINIO=$(host $DOMINIO | grep "has address" | head -1 | awk '{print $4}')
fi

if [ -n "$IP_DOMINIO" ]; then
    echo "IP do domínio: $IP_DOMINIO"
    echo ""
    
    # Verificar PTR (reverse DNS)
    if [ "$DNS_TOOL" = "dig" ]; then
        PTR_RECORD=$(dig +short -x $IP_DOMINIO 2>/dev/null)
    else
        PTR_RECORD=$(host $IP_DOMINIO 2>/dev/null | grep "domain name pointer" | awk '{print $5}')
    fi
    
    if [ -n "$PTR_RECORD" ]; then
        echo -e "${VERDE}✓ PTR/rDNS ENCONTRADO${NC}"
        echo "  IP: $IP_DOMINIO"
        echo "  PTR: $PTR_RECORD"
        echo ""
        
        # Verificar se o PTR aponta de volta para o domínio (forward-confirmed reverse DNS)
        if echo "$PTR_RECORD" | grep -q "$DOMINIO"; then
            echo -e "${VERDE}  ✓ PTR corresponde ao domínio (FCrDNS válido)${NC}"
        else
            echo -e "${AMARELO}  ! PTR não corresponde exatamente ao domínio${NC}"
            echo "    Isso pode afetar a entrega de emails"
        fi
    else
        echo -e "${VERMELHO}✗ PTR/rDNS NÃO CONFIGURADO${NC}"
        echo "  IP: $IP_DOMINIO"
        echo ""
        echo "  IMPORTANTE: A falta de PTR afeta a entrega de emails!"
        echo "  O PTR deve ser configurado no provedor de hospedagem/VPS"
        echo "  (não é configurado no domínio, mas no IP)"
    fi
else
    echo -e "${VERMELHO}✗ Não foi possível resolver o IP do domínio${NC}"
fi

echo ""

# ============================================================
# RESUMO FINAL
# ============================================================
echo "============================================================"
echo "RESUMO DA VERIFICAÇÃO"
echo "============================================================"

# Contar verificações
TOTAL=0
OK=0
ALERTA=0
ERRO=0

# SPF
TOTAL=$((TOTAL + 1))
if [ -n "$SPF_RECORD" ]; then
    if echo "$SPF_RECORD" | grep -q "v=spf1.*all"; then
        OK=$((OK + 1))
        echo -e "${VERDE}✓ SPF${NC} - Configurado corretamente"
    else
        ALERTA=$((ALERTA + 1))
        echo -e "${AMARELO}! SPF${NC} - Configurado mas com problemas"
    fi
else
    ERRO=$((ERRO + 1))
    echo -e "${VERMELHO}✗ SPF${NC} - Ausente"
fi

# DMARC
TOTAL=$((TOTAL + 1))
if [ -n "$DMARC_RECORD" ]; then
    if echo "$DMARC_RECORD" | grep -q "v=DMARC1.*p="; then
        OK=$((OK + 1))
        echo -e "${VERDE}✓ DMARC${NC} - Configurado corretamente"
    else
        ALERTA=$((ALERTA + 1))
        echo -e "${AMARELO}! DMARC${NC} - Configurado mas com problemas"
    fi
else
    ERRO=$((ERRO + 1))
    echo -e "${VERMELHO}✗ DMARC${NC} - Ausente"
fi

# DKIM
TOTAL=$((TOTAL + 1))
if [ $DKIM_ENCONTRADO -eq 1 ]; then
    OK=$((OK + 1))
    echo -e "${VERDE}✓ DKIM${NC} - Configurado"
else
    ALERTA=$((ALERTA + 1))
    echo -e "${AMARELO}! DKIM${NC} - Não encontrado (pode usar seletor personalizado)"
fi

# PTR
TOTAL=$((TOTAL + 1))
if [ -n "$PTR_RECORD" ]; then
    OK=$((OK + 1))
    echo -e "${VERDE}✓ PTR/rDNS${NC} - Configurado"
else
    ERRO=$((ERRO + 1))
    echo -e "${VERMELHO}✗ PTR/rDNS${NC} - Ausente"
fi

echo ""
echo "============================================================"
echo "TOTAL: $TOTAL verificações"
echo -e "  ${VERDE}OK: $OK${NC}"
echo -e "  ${AMARELO}Alertas: $ALERTA${NC}"
echo -e "  ${VERMELHO}Erros: $ERRO${NC}"
echo "============================================================"

if [ $ERRO -gt 0 ]; then
    echo ""
    echo -e "${VERMELHO}AÇÕES NECESSÁRIAS:${NC}"
    echo "  - Configure os registros ausentes para melhorar a entrega de emails"
    echo "  - SPF e DMARC são essenciais para evitar que emails caiam em spam"
    echo "  - PTR deve ser configurado no provedor de hospedagem"
fi

if [ $ALERTA -gt 0 ]; then
    echo ""
    echo -e "${AMARELO}MELHORIAS RECOMENDADAS:${NC}"
    echo "  - Revise as configurações marcadas com alerta"
    echo "  - Use políticas mais rigorosas (-all em vez de ~all no SPF)"
    echo "  - Configure DMARC com p=reject após testar com p=none"
fi

echo ""
echo "============================================================"
echo "Fim do relatório"
echo "============================================================"
